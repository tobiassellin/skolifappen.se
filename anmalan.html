🔐 Snabbguide: Konton & Firestore Security Rules

Lägg upp dessa 3 användare i Firebase Console → Authentication → Users (med valfria lösenord som du skickar till respektive distriktsledare):

  • Jämtland / Härjedalen: jamtland@skolifwebben.se
  • Medelpad: medelpad@skolifwebben.se
  • Ångermanland: angermanland@skolifwebben.se
  • Admin (Du): tobias.sellin@gmail.com (via Google-inloggning)

Lägg till följande block inuti match /databases/{database}/documents { ... } under Firestore Database → Rules: 

// UTBILDNINGSANMÄLAN & HOTELLBOKNING (3 DISTRIKT + ADMIN + AUTOMATISKT LÅS)
match /utbildning_anmalan/{docId} {
  function userEmail() {
    return (request.auth != null && 'email' in request.auth.token && request.auth.token.email != null)
      ? request.auth.token.email.lower()
      : '';
  }
  function isAdmin() {
    let em = userEmail();
    return em == 'tobias.sellin@gmail.com' || em == 'admin@skolifwebben.se';
  }
  function isOwnDistrict() {
    let em = userEmail();
    return (docId == 'jamtland' && em == 'jamtland@skolifwebben.se') ||
           (docId == 'medelpad' && em == 'medelpad@skolifwebben.se') ||
           (docId == 'angermanland' && em == 'angermanland@skolifwebben.se');
  }
  function isLocked() {
    let cfgPath = /databases/$(database)/documents/utbildning_anmalan/_config;
    return exists(cfgPath) && (
      get(cfgPath).data.get('isLocked', false) == true ||
      (
        get(cfgPath).data.get('overrideOpenAfterDeadline', false) != true &&
        get(cfgPath).data.get('deadlineTs', null) is timestamp &&
        request.time >= get(cfgPath).data.get('deadlineTs', null)
      )
    );
  }
  allow read: if isAdmin() || isOwnDistrict() || (request.auth != null && docId == '_config');
  allow write: if isAdmin() || (isOwnDistrict() && !isLocked());
}
